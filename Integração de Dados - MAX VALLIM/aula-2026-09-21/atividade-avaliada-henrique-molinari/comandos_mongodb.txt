// 3.1 Listar todos os produtos disponiveis de uma categoria

// 3.2 Historico de compras de um cliente especifico (C1 - Ana Beatriz Souza)

db.pedidos.find({ id_cliente: "C1" })

db.clientes.aggregate([
  { $match: { _id: "C1" } },
  { $lookup: { from: "pedidos", localField: "_id", foreignField: "id_cliente", as: "compras" } },
  { $project: { nome_completo: 1, email: 1, compras: 1 } }
])
